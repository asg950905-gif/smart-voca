```python
# src/ui/tabs/study_tab.py

class StudyTab(QWidget):
    # Signals
    progress_updated = pyqtSignal(int, int)  # (current, total)
    study_completed = pyqtSignal()
    word_changed = pyqtSignal(str, str)      # (english, korean)
    
    def __init__(self, study_manager):
        super().__init__()
        self.study_manager = study_manager
        self.current_word_index = 0
        self.daily_results = []  # [True, False, True, ...]
        
        self.init_ui()
        self.load_today_words()
    
    def init_ui(self):
        """UI 생성"""
        layout = QVBoxLayout()
        
        # 진도 정보 영역
        progress_group = self.create_progress_group()
        layout.addWidget(progress_group)
        
        # 단어 표시 영역
        word_group = self.create_word_group()
        layout.addWidget(word_group)
        
        # 성적 표시 영역
        stats_group = self.create_stats_group()
        layout.addWidget(stats_group)
        
        # 남은 단어 테이블
        self.table_widget = self.create_word_table()
        layout.addWidget(self.table_widget)
        
        layout.setContentsMargins(20, 20, 20, 20)
        layout.setSpacing(15)
        self.setLayout(layout)
    
    def create_progress_group(self):
        """진도 정보 그룹박스"""
        group = QGroupBox("학습 진도")
        layout = QVBoxLayout()
        
        self.label_progress = QLabel()
        self.label_progress.setFont(QFont("Arial", 12, QFont.Bold))
        
        self.progress_bar = QProgressBar()
        self.progress_bar.setValue(0)
        
        self.label_daily = QLabel()
        
        layout.addWidget(self.label_progress)
        layout.addWidget(self.progress_bar)
        layout.addWidget(self.label_daily)
        
        group.setLayout(layout)
        return group
    
    def create_word_group(self):
        """단어 표시 그룹박스"""
        group = QGroupBox("오늘의 단어")
        layout = QVBoxLayout()
        layout.setAlignment(Qt.AlignmentFlag.AlignCenter)
        
        # 영단어 레이블
        self.label_english = QLabel()
        font_en = QFont("Arial", 36, QFont.Bold)
        self.label_english.setFont(font_en)
        self.label_english.setAlignment(Qt.AlignmentFlag.AlignCenter)
        
        # 한글 레이블
        self.label_korean = QLabel()
        font_ko = QFont("맑은 고딕", 24)
        self.label_korean.setFont(font_ko)
        self.label_korean.setAlignment(Qt.AlignmentFlag.AlignCenter)
        
        # 버튼 (수평 레이아웃)
        button_layout = QHBoxLayout()
        button_layout.setSpacing(20)
        
        self.btn_ok = QPushButton("✓ 외웠어요!")
        self.btn_ok.setFixedSize(150, 50)
        self.btn_ok.setStyleSheet("background-color: #28a745; color: white; font-weight: bold;")
        self.btn_ok.clicked.connect(self.on_ok_clicked)
        
        self.btn_x = QPushButton("✗ 아직 못 외웠어요")
        self.btn_x.setFixedSize(150, 50)
        self.btn_x.setStyleSheet("background-color: #dc3545; color: white; font-weight: bold;")
        self.btn_x.clicked.connect(self.on_x_clicked)
        
        button_layout.addStretch()
        button_layout.addWidget(self.btn_ok)
        button_layout.addWidget(self.btn_x)
        button_layout.addStretch()
        
        layout.addStretch()
        layout.addWidget(self.label_english)
        layout.addWidget(self.label_korean)
        layout.addSpacing(30)
        layout.addLayout(button_layout)
        layout.addStretch()
        
        group.setLayout(layout)
        return group
    
    def create_stats_group(self):
        """성적 표시 그룹박스"""
        group = QGroupBox("오늘 성적")
        layout = QVBoxLayout()
        
        self.label_stats = QLabel()
        self.label_stats.setFont(QFont("Arial", 10))
        
        self.stats_progress = QProgressBar()
        
        layout.addWidget(self.label_stats)
        layout.addWidget(self.stats_progress)
        
        group.setLayout(layout)
        return group
    
    def create_word_table(self):
        """남은 단어 테이블"""
        table = QTableWidget()
        table.setColumnCount(4)
        table.setHorizontalHeaderLabels(["#", "영단어", "한글", "상태"])
        table.setColumnWidth(0, 30)
        table.setColumnWidth(1, 150)
        table.setColumnWidth(2, 150)
        table.setColumnWidth(3, 100)
        return table
    
    def load_today_words(self):
        """오늘 학습할 단어 로드"""
        self.today_words = self.study_manager.get_daily_words()
        self.current_word_index = 0
        self.daily_results = []
        self.display_word()
    
    def display_word(self):
        """현재 단어 표시"""
        if self.current_word_index < len(self.today_words):
            word = self.today_words[self.current_word_index]
            self.label_english.setText(word.english)
            self.label_korean.setText(word.korean)
            self.word_changed.emit(word.english, word.korean)
            
            # 진도 업데이트
            self.update_progress_display()
        else:
            self.on_study_completed()
    
    def on_ok_clicked(self):
        """O 버튼 클릭"""
        self.daily_results.append(True)
        self.current_word_index += 1
        self.display_word()
    
    def on_x_clicked(self):
        """X 버튼 클릭"""
        self.daily_results.append(False)
        self.current_word_index += 1
        self.display_word()
    
    def update_progress_display(self):
        """진도 정보 표시 업데이트"""
        day = self.study_manager.progress.current_day
        total = self.study_manager.progress.total_days
        current_count = self.current_word_index
        total_count = len(self.today_words)
        
        self.label_progress.setText(f"{day}일차 / {total}일차")
        self.progress_bar.setValue(int((day / total) * 100))
        self.label_daily.setText(f"오늘 학습: {total_count}단어 ({current_count}/{total_count})")
        
        # 성공률 계산
        if self.current_word_index > 0:
            success = sum(self.daily_results)
            rate = (success / self.current_word_index) * 100
            self.label_stats.setText(f"완료: {self.current_word_index}/{total_count}  성공률: {rate:.0f}%")
            self.stats_progress.setValue(int((self.current_word_index / total_count) * 100))
        
        self.progress_updated.emit(day, total)
    
    def on_study_completed(self):
        """학습 완료"""
        # 1. StudyManager에 결과 기록
        self.study_manager.record_study(
            self.today_words,
            self.daily_results
        )
        
        # 2. 신호 발생
        self.study_completed.emit()
        
        # 3. 완료 메시지 표시
        QMessageBox.information(
            self,
            "학습 완료",
            "오늘 학습을 완료했습니다!\n내일 또 만나요 💪"
        )
        
        # 4. UI 초기화
        self.load_today_words()
```

```python
def weighted_problem_selection(count: int):
    """
    가중치 기반 문제 선택 (v1.0 재사용)
    
    가중치 계산:
    - 약점 단어: 50%
    - 미학습 단어: 30%
    - 전체 단어: 20%
    """
    
    weak_words = get_weak_words()  # 약점 단어
    unlearned = get_unlearned_words()  # 미학습
    all_words = get_all_words()  # 전체
    
    weights = {
        'weak': 0.5,
        'unlearned': 0.3,
        'all': 0.2
    }
    
    selected = []
    
    # 약점 단어: 50%
    weak_count = int(count * weights['weak'])
    selected.extend(random.sample(weak_words, min(weak_count, len(weak_words))))
    
    # 미학습 단어: 30%
    unlearned_count = int(count * weights['unlearned'])
    available = [w for w in unlearned if w not in selected]
    selected.extend(random.sample(available, min(unlearned_count, len(available))))
    
    # 전체 단어: 20%
    remaining = count - len(selected)
    available = [w for w in all_words if w not in selected]
    selected.extend(random.sample(available, min(remaining, len(available))))
    
    return selected[:count]
```

```python
def grade_answer(user_answer, correct, question_format):
    """
    답변 채점
    """
    
    if question_format == '4지선다형':
        return user_answer == correct
    
    elif question_format == '단답형':
        # 띄어쓰기 무시, 대소문자 무시
        user_normalized = user_answer.strip().lower()
        correct_normalized = correct.strip().lower()
        return user_normalized == correct_normalized
    
    elif question_format == 'OX형':
        # O 또는 X
        return user_answer in ['O', 'X'] and user_answer == correct
    
    return False
```
